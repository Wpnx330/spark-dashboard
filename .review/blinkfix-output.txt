// Pad data to a fixed number of points so the SVG path always has the same
// number of commands, enabling smooth CSS `d` transitions between frames.
const CHART_POINTS = 60

export function padData(data: DataPoint[], maxPoints = CHART_POINTS, pad = true): DataPoint[] {
  if (data.length === 0) return []

  // Downsample when exceeding maxPoints (always useful)
  if (data.length > maxPoints) {
    const stride = Math.ceil(data.length / maxPoints)
    const sampled: DataPoint[] = []
    for (let i = 0; i < data.length; i += stride) {
      sampled.push(data[i])
    }
    if (sampled[sampled.length - 1] !== data[data.length - 1]) {
      sampled.push(data[data.length - 1])
    }
    return sampled
  }

  // Pad only in buffer mode (live streaming needs fixed point count for
  // smooth CSS path transitions). In history mode, render data as-is so
  // the x-axis reflects the actual time range, not padded with fake points.

/** Exactly 3 ticks — start, mid, end (Chris's preference: dense time axes
 * are unreadable). End == "now" when callers pass now-anchored domains.
 * Ranges >= 4h round to the hour (history); shorter ranges stay exact
 * (seconds) so live chart end/ start labels match wall-clock. */
function generateTicks(min: number, max: number): number[] {
  const range = max - min
  const HOUR = 3_600_000
  const round = range >= 4 * HOUR
  return [0, 0.5, 1].map((f) => {
    const raw = min + range * f
    return round ? Math.round(raw / HOUR) * HOUR : Math.round(raw / 1000) * 1000
  })
}
